create table public.portfolio_holdings (
  id bigserial not null,
  symbol character varying(50) not null,
  currency character varying(10) not null default 'THB'::character varying,
  volume numeric(18, 8) not null default 0,
  initial_cost numeric(18, 4) not null default 0,
  present_price numeric(18, 4) not null default 0,
  exchange_rate numeric(10, 4) not null default 1.0,
  total_cost numeric GENERATED ALWAYS as ((volume * initial_cost)) STORED (18, 4) null,
  total_cost_thb numeric GENERATED ALWAYS as (
    case
      when ((currency)::text = 'USD'::text) then ((volume * initial_cost) * exchange_rate)
      else (volume * initial_cost)
    end
  ) STORED (18, 4) null,
  total_present_price numeric GENERATED ALWAYS as ((volume * present_price)) STORED (18, 4) null,
  total_present_price_thb numeric GENERATED ALWAYS as (
    case
      when ((currency)::text = 'USD'::text) then ((volume * present_price) * exchange_rate)
      else (volume * present_price)
    end
  ) STORED (18, 4) null,
  yield_percent numeric GENERATED ALWAYS as (
    case
      when (initial_cost > (0)::numeric) then (
        ((present_price - initial_cost) / initial_cost) * (100)::numeric
      )
      else (0)::numeric
    end
  ) STORED (10, 4) null,
  updated_at timestamp with time zone null default now(),
  user_id uuid null default auth.uid (),
  constraint portfolio_holdings_pkey primary key (id),
  constraint portfolio_holdings_symbol_key unique (symbol),
  constraint portfolio_holdings_user_id_fkey foreign KEY (user_id) references auth.users (id) on delete CASCADE
) TABLESPACE pg_default;

create table public.trade_transactions (
  id bigserial not null,
  broker character varying(30) not null,
  order_id character varying(50) null,
  trade_date date not null,
  trade_year integer GENERATED ALWAYS as (
    EXTRACT(
      year
      from
        trade_date
    )
  ) STORED null,
  side character varying(10) not null,
  stock_symbol character varying(50) not null,
  currency character varying(10) not null default 'THB'::character varying,
  units numeric(18, 8) not null,
  unit_price numeric(18, 4) not null,
  exchange_rate numeric(12, 6) null default 1.0,
  gross_amount numeric(18, 4) null,
  fee numeric(12, 4) null default 0.00,
  withholding_tax numeric(12, 4) null default 0.00,
  net_amount numeric(18, 4) not null,
  gross_amount_thb numeric(18, 4) null,
  fee_thb numeric(12, 4) null default 0.00,
  net_amount_thb numeric(18, 4) not null,
  reason text null,
  created_at timestamp with time zone null default now(),
  user_id uuid null default auth.uid (),
  constraint trade_transactions_pkey primary key (id),
  constraint trade_transactions_user_id_fkey foreign KEY (user_id) references auth.users (id) on delete CASCADE,
  constraint trade_transactions_side_check check (
    (
      (side)::text = any (
        (
          array[
            'BUY'::character varying,
            'SELL'::character varying,
            'FREE'::character varying
          ]
        )::text[]
      )
    )
  )
) TABLESPACE pg_default;

create index IF not exists idx_trades_broker on public.trade_transactions using btree (broker) TABLESPACE pg_default;

create index IF not exists idx_trades_symbol on public.trade_transactions using btree (stock_symbol) TABLESPACE pg_default;

create index IF not exists idx_trades_date on public.trade_transactions using btree (trade_date desc) TABLESPACE pg_default;
